# ─── Authoring a theme ────────────────────────────────────────────────
#
# 1. Copy this file to `themes/<your-theme-name>.toml` in your config
#    directory.
# 2. Uncomment the `[palette]` lines you want to change and edit their
#    values.  In most cases that's all you need — every style below
#    derives from the palette at load time and will track your new
#    shades automatically.  Any palette entry left commented falls
#    through to the compiled-in default.
# 3. Optionally, override individual styles by filling in the matching
#    `[h1]`, `[modal_input_focused]`, etc. table below.  The headers
#    are listed bare (no `fg` / `bg` / modifier fields) precisely so
#    you can see the full set of override slots and add fields under
#    whichever you want to customise.  Anything you leave empty falls
#    through to the palette-derived default.
# 4. Activate the theme by setting `theme = "<your-theme-name>"` in
#    `config.toml`.
#
# Colour values accept named CSS-ish colours (`"orange"`), six-digit
# hex (`"#ff8c3c"`), or 256-colour palette indices either as a string
# (`"236"`) or a bare integer (`236`).  Modifier flags (`bold`,
# `italic`, `underlined`, `reversed`, `crossed_out`, `dim`) accept
# `true` / `false` and combine freely.
#
# ─── Style merge order ────────────────────────────────────────────────
#
# At load time edamame runs a three-stage merge:
#
#   1. Start from the compiled-in default palette.
#   2. Apply your `[palette]` overrides on top.
#   3. Build a fresh style table from the merged palette, then apply
#      any per-element style overrides from this file.
#
# An empty style section (e.g. just `[h1]` with no fields) is treated
# as "no override" — the palette-derived default wins.  That's why the
# default ships with empty headers: they are intentionally palette-
# only, so editing the palette retints the whole UI without the per-
# element sections fighting back.
#
# ─── Per-element override reference ──────────────────────────────────
#
# Every style table accepts the same set of fields.  You can mix and
# match — anything you omit stays at its palette-derived value.
#
#   [some_element]
#   fg          = "#e6e6e6"   # foreground colour
#   bg          = "#121216"   # background colour
#   bold        = true        # SGR bold attribute
#   italic      = true        # SGR italic
#   underlined  = true        # SGR underline
#   reversed    = true        # swap fg/bg at render time
#   crossed_out = true        # strikethrough
#   dim         = true        # SGR faint
#
# Colour values accept any of:
#
#   fg = "yellow"        # ANSI-named colour (case-insensitive).
#                        # The 16-colour names are: black, red, green,
#                        # yellow, blue, magenta, cyan, gray,
#                        # darkgray, lightred, lightgreen,
#                        # lightyellow, lightblue, lightmagenta,
#                        # lightcyan, white.
#   fg = "#ff8c3c"       # six-digit hex
#   fg = "236"           # 256-colour palette index, as a string
#   bg = 236             # 256-colour palette index, as a bare integer
#   fg = "Reset"         # explicit terminal default — punches through
#                        # the theme's `default_text` / `default_bg`
#                        # so the underlying terminal colour shows.
#                        # Useful on `[normal]` for users who want
#                        # edamame to inherit their terminal's bg.
#
# Worked example — make H1 a green underlined heading without any
# bold weight, and give the inline-code span an explicit dark-grey
# background regardless of palette:
#
#   [h1]
#   fg         = "#9ad26e"
#   underlined = true
#   bold       = false       # (also the default; shown for clarity)
#
#   [code_span]
#   fg = "lightyellow"
#   bg = "#1a1a1f"
#
# Worked example — opt out of the theme's coloured document surface
# so edamame inherits whatever bg your terminal is using:
#
#   [normal]
#   fg = "Reset"
#   bg = "Reset"
#
# Worked example — give the focused mode chip a louder treatment by
# stacking modifiers:
#
#   [status_mode_rendered]
#   fg          = "#121216"
#   bg          = "#ff963c"
#   bold        = true
#   reversed    = true
